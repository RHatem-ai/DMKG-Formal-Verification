lemma secrecy_vmss_contributions:
  all-traces
  "∀ P X1 X2 Y1 Y2 #i.
    (VMSSSecretContributionsGenerated( P, X1, X2, Y1, Y2 ) @ #i) ⇒
    (¬((((∃ #j. K( X1 ) @ #j) ∨ (∃ #j. K( X2 ) @ #j)) ∨
        (∃ #j. K( Y1 ) @ #j)) ∨
       (∃ #j. K( Y2 ) @ #j)))"
/*
guarded formula characterizing all counter-examples:
"∃ P X1 X2 Y1 Y2 #i.
  (VMSSSecretContributionsGenerated( P, X1, X2, Y1, Y2 ) @ #i)
 ∧
  ((∃ #j. (K( X1 ) @ #j)) ∨
   (∃ #j. (K( X2 ) @ #j)) ∨
   (∃ #j. (K( Y1 ) @ #j)) ∨
   (∃ #j. (K( Y2 ) @ #j)))"
*/
simplify
solve( (∃ #j. (K( ~x1i ) @ #j))  ∥ (∃ #j. (K( ~x2i ) @ #j))  ∥
       (∃ #j. (K( ~y1i ) @ #j))  ∥ (∃ #j. (K( ~y2i ) @ #j)) )
  case case_1
  solve( AuthorizeVMSSTranscript( P ) ▶₀ #i )
    case RegisterParticipant
    solve( !HonestParticipant( ~ident ) ▶₁ #i )
      case ActivateHonestParticipant
      by solve( !KU( ~x1i ) @ #vk )
    qed
  qed
next
  case case_2
  solve( AuthorizeVMSSTranscript( P ) ▶₀ #i )
